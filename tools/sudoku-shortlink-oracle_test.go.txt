func TestSakuraShortLinkGoldenVectors(t *testing.T) {
	output := os.Getenv("SUDOKU_URI_GOLDEN_OUTPUT")
	if output == "" {
		t.Skip("golden output not requested")
	}
	var rows []map[string]any
	inputs := []string{
		`{"name":"香港 日本 + /?#% 😀","mode":"client","server_address":"[2001:db8::1]:443","local_port":10233,"key":"synthetic-key","aead":"aes-128-gcm","ascii":"prefer_ascii","padding_min":5,"padding_max":15,"enable_pure_downlink":false,"custom_tables":["xpxvvpvv","vxpvxvvp"],"httpmask":{"disable":false,"mode":"auto","tls":true,"host":"cdn.example.com","path_root":"root_01","multiplex":"on"}}`,
		`{"mode":"client","server_address":"edge.example.com:8443","local_port":1081,"key":"synthetic-key","aead":"chacha20-poly1305","ascii":"up_ascii_down_entropy","padding_min":5,"padding_max":15,"enable_pure_downlink":true,"custom_table":"xpxvvpvv","httpmask":{"disable":true,"mode":"legacy","multiplex":"on"}}`,
	}
	for _, mode := range []string{"legacy", "stream", "poll", "auto", "ws"} {
		inputs = append(inputs, `{"mode":"client","server_address":"example.com:443","key":"synthetic-key","aead":"chacha20-poly1305","ascii":"prefer_entropy","padding_min":5,"padding_max":15,"enable_pure_downlink":true,"httpmask":{"mode":"`+mode+`","multiplex":"auto"}}`)
	}
	for _, input := range inputs {
		var cfg Config
		if err := json.Unmarshal([]byte(input), &cfg); err != nil {
			t.Fatal(err)
		}
		if err := cfg.Finalize(); err != nil {
			t.Fatal(err)
		}
		link, err := BuildShortLinkFromConfig(&cfg, "")
		if err != nil {
			t.Fatal(err)
		}
		decoded, err := BuildConfigFromShortLink(link)
		if err != nil {
			t.Fatal(err)
		}
		rows = append(rows, map[string]any{"link": link, "config": decoded})
	}
	data, err := json.MarshalIndent(rows, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(output, append(data, '\n'), 0600); err != nil {
		t.Fatal(err)
	}
}

func TestSakuraExportedShortLinks(t *testing.T) {
	directory := os.Getenv("SUDOKU_URI_JVM_FIXTURES")
	if directory == "" {
		t.Skip("JVM fixtures not requested")
	}
	entries, err := filepath.Glob(filepath.Join(directory, "*.txt"))
	if err != nil || len(entries) == 0 {
		t.Fatalf("missing fixtures: %v", err)
	}
	for _, entry := range entries {
		t.Run(filepath.Base(entry), func(t *testing.T) {
			link, err := os.ReadFile(entry)
			if err != nil {
				t.Fatal(err)
			}
			cfg, err := BuildConfigFromShortLink(string(link))
			if err != nil {
				t.Fatal(err)
			}
			raw, err := os.ReadFile(strings.TrimSuffix(entry, ".txt") + ".json")
			if err != nil {
				t.Fatal(err)
			}
			var out struct {
				Key        string          `json:"key"`
				AEAD       string          `json:"aead"`
				ASCII      string          `json:"ascii"`
				PaddingMin int             `json:"padding_min"`
				PaddingMax int             `json:"padding_max"`
				Pure       bool            `json:"enable_pure_downlink"`
				HTTPMask   *HTTPMaskConfig `json:"httpmask"`
			}
			if err := json.Unmarshal(raw, &out); err != nil {
				t.Fatal(err)
			}
			if cfg.Key != strings.TrimSpace(out.Key) || cfg.AEAD != out.AEAD || cfg.ASCII != normalizeASCII(out.ASCII) ||
				cfg.PaddingMin != out.PaddingMin || cfg.PaddingMax != out.PaddingMax || cfg.EnablePureDownlink != out.Pure {
				t.Fatal("outbound/official parser mismatch")
			}
			if out.HTTPMask != nil {
				expected := *out.HTTPMask
				if cfg.HTTPMask.Disable != expected.Disable || cfg.HTTPMask.TLS != expected.TLS ||
					cfg.HTTPMask.Mode != normalizeHTTPMaskMode(expected.Mode) ||
					cfg.HTTPMask.Host != strings.TrimSpace(expected.Host) || cfg.HTTPMask.PathRoot != strings.TrimSpace(expected.PathRoot) ||
					cfg.HTTPMask.Multiplex != normalizeMultiplex(expected.Multiplex) {
					t.Fatal("HTTPMask mapping mismatch")
				}
			}
		})
	}
}
